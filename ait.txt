## 1. Genetic Algorithm (maximize number of 1s)

```python
import random
pop=[[random.randint(0,1) for _ in range(8)] for _ in range(10)]
fit=lambda g:sum(g)
for _ in range(15):
    pop.sort(key=fit,reverse=True)
    elite=pop[:4]
    cross=[[random.choice(a+b) for a,b in zip(random.choice(elite),random.choice(elite))] for _ in range(4)]
    mut=[[1-x if random.random()<.1 else x for x in random.choice(elite)] for _ in range(2)]
    pop=elite+cross+mut
print(max(pop,key=fit))
```

## 2. Alpha-Beta Pruning (tic-tac-toe)

```python
def win(b):
    for i,j,k in [(0,1,2),(3,4,5),(6,7,8),(0,3,6),(1,4,7),(2,5,8),(0,4,8),(2,4,6)]:
        if b[i]==b[j]==b[k]!=' ': return b[i]
    return None

def ab(b,p='X',a=-2,beta=2):
    w=win(b)
    if w=='X': return 1
    if w=='O': return -1
    if ' ' not in b: return 0
    if p=='X':
        v=-2
        for i in range(9):
            if b[i]==' ':
                v=max(v,ab(b[:i]+'X'+b[i+1:],'O',a,beta)); a=max(a,v)
                if a>=beta: break
        return v
    v=2
    for i in range(9):
        if b[i]==' ':
            v=min(v,ab(b[:i]+'O'+b[i+1:],'X',a,beta)); beta=min(beta,v)
            if a>=beta: break
    return v
```

## 3. Tic-Tac-Toe using Minimax

```python
def win(b):
    for i,j,k in [(0,1,2),(3,4,5),(6,7,8),(0,3,6),(1,4,7),(2,5,8),(0,4,8),(2,4,6)]:
        if b[i]==b[j]==b[k]!=' ': return b[i]
    return None

def mm(b,p='X'):
    w=win(b)
    if w=='X': return 1
    if w=='O': return -1
    if ' ' not in b: return 0
    vals=[mm(b[:i]+p+b[i+1:],'O' if p=='X' else 'X') for i in range(9) if b[i]==' ']
    return max(vals) if p=='X' else min(vals)

b='XO X O   '
best=max((mm(b[:i]+'X'+b[i+1:],'O'),i) for i in range(9) if b[i]==' ')
print("Best move:",best[1])
```

## 4. Water Jug Problem (BFS)

```python
from collections import deque
def jug(a,b,t):
    q=deque([(0,0)]); seen={(0,0)}
    while q:
        x,y=q.popleft()
        if t in (x,y): return True
        for s in [(a,y),(x,b),(0,y),(x,0),
                  (max(0,x-(b-y)),min(b,x+y)),
                  (min(a,x+y),max(0,y-(a-x)))]:
            if s not in seen: seen.add(s); q.append(s)
    return False

print(jug(4,3,2))   # True
```

## 5. Missionaries and Cannibals (BFS)

```python
from collections import deque
def mc():
    q=deque([(3,3,1)]); seen={(3,3,1)}
    while q:
        m,c,b=q.popleft()
        if (m,c)==(0,0): return True
        for dm,dc in [(1,0),(2,0),(0,1),(0,2),(1,1)]:
            nm,nc=m-dm*(1 if b else -1), c-dc*(1 if b else -1)
            if 0<=nm<=3 and 0<=nc<=3 and (nm==0 or nm>=nc) and (3-nm==0 or 3-nm>=3-nc):
                s=(nm,nc,1-b)
                if s not in seen: seen.add(s); q.append(s)
    return False

print(mc())   # True
```

## 6. 8-Puzzle (BFS)

```python
from collections import deque
def solve8(start,goal=(1,2,3,4,5,6,7,8,0)):
    q=deque([(start,0)]); seen={start}
    while q:
        s,d=q.popleft()
        if s==goal: return d
        i=s.index(0)
        for j in (i-3,i+3,i-1,i+1):
            if 0<=j<9 and (abs(i-j)!=1 or i//3==j//3):
                t=list(s); t[i],t[j]=t[j],t[i]; t=tuple(t)
                if t not in seen: seen.add(t); q.append((t,d+1))
    return -1

print(solve8((1,2,3,4,5,6,0,7,8)))   # 2
```

## 7. 8-Queens (Backtracking)

```python
def queens(n=8,r=0,pos=()):
    if r==n: return pos
    for c in range(n):
        if all(c!=pc and abs(c-pc)!=r-pr for pr,pc in enumerate(pos)):
            res=queens(n,r+1,pos+(c,))
            if res: return res
    return None

print(queens())   # (0,4,7,5,2,6,1,3) etc.
```
